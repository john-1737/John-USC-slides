\subsection{Transaction Costs}
\textbf{Transaction Costs}
\begin{itemize}
\item The time and money spent in carrying out financial transactions
\end{itemize}
\textbf{Risk Sharing }
\begin{itemize}
\item Create and sell assets with risk characteristics that investors want, and buy assets with risk characteristics that issuers want.   Sometimes this means more risk, sometimes it means less risk.   Sometimes it means different risk.
\end{itemize}
Information Costs
\begin{itemize}
\item Financial transactions are often characterized by asymmetric information -- that is one party has information that the other party does not.   For example, a borrower has better information about ability to repay than the lender.
\item Adverse selection -- asymmetric information \textit{before} the transaction occurs
\item Moral hazard -- asymmetric information \textit{after} the transaction occurs
\end{itemize}

\section{Lecture 3: Market Returns}

\subsection{Total Returns}
    \begin{shaded}
    \subsection{Practice: Dividends}
     Stock X just paid a dividend of \$10.  You then buy Stock X for \$1000, hold it for one year during which time it pays 4 more dividends of the same amount.   After one year you then sell the stock for \$1100.  What was your total return?\\
     \vspace{1em}
    $$ \text{Total Return} = \frac{4\times10 + 10}{1000} = 50\% $$
    
    \end{shaded}

    \begin{shaded}
        \subsection{Practice: More Total Returns}
        \begin{enumerate}
        \item Based on the information below, what is the total return for an investor in Coca-Cola (KO) that purchased the stock on Jan 1, 2025 and sold it one year later.\\
        Price:
            \begin{itemize}
                \item Jan 1, 2025:  \$61.87
                \item Dec 31, 2025: \$69.93
            \end{itemize}
    	Dividends: 
            \begin{itemize}
                \item \$0.51 per quarter 
            \end{itemize}

            \begin{align*}
            \text{Total Return} &= \frac{(69.93-61.87) + 0.51\dot4}{61.87} = 16\%\\
            \text{Capital Gain} &= \frac{(69.93-61.87)}{61.87} = 13\%\\
            \text{Income Return} &= \frac{0.51\dot4}{61.87} = 3\%
            \end{align*}
    
        \item Consider a 10-year Treasury bond that has a 4.25\% coupon, which is paid semi-annually and is based on the face value of the bond (i.e. \$100).   An investor purchases this bond for \$97.26 on Jan 1, 2025.   She sells the bond one year later and realizes a total return of 8.2\%.   What price did she sell the bond for?

        \begin{align*}
            &\text{Total Return} = 8.2\% = \frac{\text{Sale Price} - 97.26 + 4.25}{97.26} \\
            &0.082 \times 97.26 = \text{Sale Price} - 97.26 + 4.25 \\
            &0.082 \times 97.26 + 97.26 - 4.25 = \text{Sale Price} \\
            &100.99 = \text{Sale Price}
        \end{align*}
        
        \end{enumerate}
    \end{shaded}

\section{Lecture 4: Bonds 1}

\subsection{Practice: Future Values and Discounting}

    \begin{shaded}
    \begin{itemize}
    \item An pension fund started with $\$1,000,000$ in 1926.   Their investments have realized an average rate of return of 7.2\% every year since.  
    \begin{itemize}
        \item Use the formula for compounding returns to determine the size of the pension today.
        \item How long will it take for the pension fund to double in value?
    \end{itemize}
    $$FV = 1m*(1.072)^{100} = 1m*1,045 \approx \$1bn$$
    Doubles approximately every 10 years
    
    \item The pension fund is considering buying a security that will be worth $10,000$ in 5 years.  The security has no risk.   The pension fund expects to continue to earn $7.2\%$ on its investments over the next 5 years.   What price would the pension fund be willing to pay for the security today? 
    $$PV = \frac{10,000}{(1+0.072)^{5}} = 7,065.5$$
    \end{itemize}
    \end{shaded}
